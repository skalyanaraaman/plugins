\newcommand{\tracefont}{\trace\fontsize{16.5pt}{\pitch}\selectfont}

\node[anchor=base east,inner sep=0pt] at (8.0in,-0.5in) {\hand\fontsize{11pt}{12pt}\selectfont\color{black!55}page \the\numexpr\value{page}-1\relax};
\fi
\end{tikzpicture}}

% art beside the text: \artbg{file}{R|L}{rows}
\newcommand{\artbg}[3]{%
\begin{tikzpicture}[remember picture,overlay,shift={(current page.north west)}]
\ifx R#2\node[anchor=north east,inner sep=0pt] at (8.2in,{-0.75in-1*\pitch}) {\includegraphics[height=#3\pitch]{art/#1.png}};\else
\node[anchor=north west,inner sep=0pt] at (1.2in,{-0.75in-1*\pitch}) {\includegraphics[height=#3\pitch]{art/#1.png}};\fi
\end{tikzpicture}}

% \entry{DAY LABEL}{art file}{R|L}{art rows}{story text = ONE paragraph, traced, wraps around the art}
\newcommand{\entry}[5]{%
\newpage\stepcounter{pg}%
\AddToShipoutPictureBG*{\artbg{#2}{#3}{#4}}%
\setlength{\artw}{\dimexpr#4\pitch*290/338\relax}%
\noindent{\marker\fontsize{17pt}{\pitch}\selectfont #1}\par
{\rightskip=0pt plus 1.6em \ifx R#3\hangindent=-\artw\else\hangindent=\artw\fi \hangafter=-#4 \noindent\dotted{\tracefont #5}\par
\ifnum\prevgraf<#4 \count255=\numexpr#4-\prevgraf\relax \vskip\count255\pitch\fi}}

% \entryd = same, but traced text on EVERY OTHER ruled line, bigger letters (ROOMY preset, about 70 words)
\newcommand{\entryd}[5]{%
\newpage\stepcounter{pg}%
\AddToShipoutPictureBG*{\artbg{#2}{#3}{#4}}%
\setlength{\artw}{\dimexpr#4\pitch*290/338\relax}%
\noindent{\marker\fontsize{17pt}{\pitch}\selectfont #1}\par
{\baselineskip=2\pitch \rightskip=0pt plus 1.6em \ifx R#3\hangindent=-\artw\else\hangindent=\artw\fi \hangafter=-\numexpr#4/2\relax \noindent\dotted{\trace\fontsize{18pt}{2\pitch}\selectfont #5}\par
\count255=\numexpr#4/2\relax \ifnum\prevgraf<\count255 \count254=\numexpr\count255-\prevgraf\relax \vskip\count254\baselineskip\fi}}

% \turnhead{prompt}, optional \starter{traced sentence start}, then \lines{n blank writing lines}
\newcommand{\turnhead}[1]{\par\vskip\pitch\noindent{\marker\fontsize{12pt}{\pitch}\selectfont YOUR TURN:}\ {\hand\fontsize{14.5pt}{\pitch}\selectfont #1}\par}
\newcommand{\starter}[1]{\par\noindent\dotted{\tracefont #1}\par}
\newcommand{\lines}[1]{\par\vskip#1\pitch}
% \drawbox{rows}{what to draw}
\newcommand{\drawbox}[2]{\par\vskip\pitch\noindent
\hbox to 0pt{\begin{tikzpicture}[overlay]
\draw[decorate,decoration={random steps,segment length=5pt,amplitude=0.5pt},line width=0.9pt,draw=black!70,fill=white,rounded corners=3pt] (-0.05in,{0.8*\pitch}) rectangle ({\textwidth+0.05in},{-#1*\pitch});
\node[anchor=base west,inner sep=0pt] at (0.12in,0) {\marker\fontsize{11pt}{13pt}\selectfont DRAW IT:\ \hand\fontsize{14pt}{16pt}\selectfont #2};
\end{tikzpicture}\hss}\par\vskip#1\pitch}
% \wizard{word}{word (noun): short meaning.}  word x3 and the meaning are both traced
